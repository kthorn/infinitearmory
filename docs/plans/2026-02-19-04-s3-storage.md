```typescript
import 'server-only'
import { z } from 'zod'

const envSchema = z.object({
  // Database
  DATABASE_URL: z.string().default('file:./dev.db'),

  // AI Providers
  TEXT_PROVIDER: z.enum(['openai', 'anthropic']).default('openai'),
  IMAGE_PROVIDER: z.enum(['openai', 'gemini']).default('openai'),
  OPENAI_API_KEY: z.string().optional(),
  ANTHROPIC_API_KEY: z.string().optional(),
  GEMINI_API_KEY: z.string().optional(),

  // S3 Storage (optional in dev — local fallback used; required in prod)
  S3_BUCKET: z.string().optional(),
  S3_REGION: z.string().default('us-east-1'),
  S3_ENDPOINT: z.string().url().optional(), // For R2 or MinIO
  S3_ACCESS_KEY_ID: z.string().optional(),
  S3_SECRET_ACCESS_KEY: z.string().optional(),
  S3_PUBLIC_URL: z.string().url().optional(), // Public URL prefix if using CDN

  // Auth
  AUTH_PASSWORD: z.string().optional(),

  // App
  NODE_ENV: z.enum(['development', 'production', 'test']).default('development'),
})

export type Env = z.infer<typeof envSchema>

function validateEnv(): Env {
  const parsed = envSchema.safeParse(process.env)

  if (!parsed.success) {
    console.error('Invalid environment variables:', parsed.error.flatten().fieldErrors)
    throw new Error('Invalid environment variables')
  }

  const data = parsed.data

  // Enforce S3 in production
  if (data.NODE_ENV === 'production') {
    if (!data.S3_BUCKET || !data.S3_ACCESS_KEY_ID || !data.S3_SECRET_ACCESS_KEY) {
      throw new Error('S3_BUCKET, S3_ACCESS_KEY_ID, and S3_SECRET_ACCESS_KEY are required in production')
    }
  }

  // If custom endpoint is set (R2/MinIO), require S3_PUBLIC_URL for correct public URLs
  if (data.S3_ENDPOINT && !data.S3_PUBLIC_URL) {
    throw new Error('S3_PUBLIC_URL is required when S3_ENDPOINT is set (R2/MinIO public URL cannot be derived from endpoint)')
  }

  return data
}

export const env = validateEnv()

// Helper to check if S3 is configured
export function isS3Configured(): boolean {
  return !!(env.S3_BUCKET && env.S3_ACCESS_KEY_ID && env.S3_SECRET_ACCESS_KEY)
}
```

```typescript
import { S3Client } from '@aws-sdk/client-s3'
import { env, isS3Configured } from '@/lib/env'

let s3Client: S3Client | null = null

export function getS3Client(): S3Client {
  if (!isS3Configured()) {
    throw new Error('S3 is not configured. Set S3_BUCKET, S3_ACCESS_KEY_ID, and S3_SECRET_ACCESS_KEY.')
  }

  if (!s3Client) {
    s3Client = new S3Client({
      region: env.S3_REGION,
      endpoint: env.S3_ENDPOINT,
      credentials: {
        accessKeyId: env.S3_ACCESS_KEY_ID!,
        secretAccessKey: env.S3_SECRET_ACCESS_KEY!,
      },
      // Required for R2 and some S3-compatible services
      forcePathStyle: !!env.S3_ENDPOINT,
    })
  }

  return s3Client
}

export function getS3Bucket(): string {
  if (!env.S3_BUCKET) {
    throw new Error('S3_BUCKET is not configured')
  }
  return env.S3_BUCKET
}
```

```typescript
import { PutObjectCommand } from '@aws-sdk/client-s3'
import { getS3Client, getS3Bucket } from './s3-client'
import { env } from '@/lib/env'

export interface UploadImageParams {
  weaponId: string
  imageData: Buffer
  assetId?: string
}

export interface UploadImageResult {
  key: string
  url: string
}

/**
 * Upload a weapon image to S3
 *
 * Key format: weapons/<weaponId>/<assetId>.png
 */
export async function uploadWeaponImage({
  weaponId,
  imageData,
  assetId,
}: UploadImageParams): Promise<UploadImageResult> {
  validateStorageId(weaponId)
  if (assetId) validateStorageId(assetId)

  const s3 = getS3Client()
  const bucket = getS3Bucket()

  // Generate asset ID if not provided (for multiple images per weapon)
  const finalAssetId = assetId ?? generateAssetId()
  const key = `weapons/${weaponId}/${finalAssetId}.png`

  await s3.send(
    new PutObjectCommand({
      Bucket: bucket,
      Key: key,
      Body: imageData,
      ContentType: 'image/png',
    })
  )

  // Generate public URL
  const url = getPublicUrl(key)

  return { key, url }
}

/**
 * Validate that an ID is safe for use in storage keys/paths
 */
function validateStorageId(id: string): void {
  if (!/^[\w-]+$/.test(id)) {
    throw new Error(`Invalid storage ID: ${id}`)
  }
}

/**
 * Generate a public URL for an S3 object
 */
function getPublicUrl(key: string): string {
  const bucket = getS3Bucket()

  // Use custom public URL if configured (for CDN or R2)
  if (env.S3_PUBLIC_URL) {
    return `${env.S3_PUBLIC_URL}/${key}`
  }

  // Standard S3 URL (S3_ENDPOINT without S3_PUBLIC_URL is rejected at startup)
  return `https://${bucket}.s3.${env.S3_REGION}.amazonaws.com/${key}`
}

/**
 * Generate a random asset ID
 */
function generateAssetId(): string {
  return `${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
}
```

```typescript
import { DeleteObjectCommand } from '@aws-sdk/client-s3'
import { getS3Client, getS3Bucket } from './s3-client'

/**
 * Delete a weapon image from S3
 */
export async function deleteWeaponImage(key: string): Promise<void> {
  const s3 = getS3Client()
  const bucket = getS3Bucket()

  await s3.send(
    new DeleteObjectCommand({
      Bucket: bucket,
      Key: key,
    })
  )
}

/**
 * Extract S3 key from a full URL
 */
export function extractKeyFromUrl(url: string): string {
  try {
    const parsed = new URL(url)
    const path = decodeURIComponent(parsed.pathname)
    const match = path.match(/weapons\/[\w-]+\/[\w.-]+\.png/)
    if (!match) {
      throw new Error(`Could not extract storage key from URL: ${url}`)
    }
    return match[0]
  } catch (e) {
    if (e instanceof TypeError) {
      throw new Error(`Invalid URL for key extraction: ${url}`)
    }
    throw e
  }
}
```

```typescript
import { mkdir, writeFile, unlink } from 'fs/promises'
import { join, resolve } from 'path'

const LOCAL_STORAGE_DIR = './public/uploads'

export interface LocalUploadParams {
  weaponId: string
  imageData: Buffer
  assetId?: string
}

export interface LocalUploadResult {
  key: string
  url: string
}

/**
 * Validate that a path stays within the storage root (prevents path traversal)
 */
function assertSafePath(filepath: string): void {
  const root = resolve(LOCAL_STORAGE_DIR)
  const resolved = resolve(filepath)
  if (!resolved.startsWith(root)) {
    throw new Error(`Path traversal detected: ${filepath}`)
  }
}

/**
 * Validate that an ID is safe for use in filesystem paths
 */
function validateId(id: string): void {
  if (!/^[\w-]+$/.test(id)) {
    throw new Error(`Invalid storage ID: ${id}`)
  }
}

/**
 * Save image to local filesystem (development fallback)
 */
export async function uploadToLocal({
  weaponId,
  imageData,
  assetId,
}: LocalUploadParams): Promise<LocalUploadResult> {
  validateId(weaponId)
  if (assetId) validateId(assetId)

  const dir = join(LOCAL_STORAGE_DIR, 'weapons', weaponId)
  assertSafePath(dir)

  await mkdir(dir, { recursive: true })

  const finalAssetId = assetId ?? `${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
  const filename = `${finalAssetId}.png`
  const filepath = join(dir, filename)
  assertSafePath(filepath)

  await writeFile(filepath, imageData)

  const key = `weapons/${weaponId}/${filename}`
  const url = `/uploads/${key}`

  return { key, url }
}

/**
 * Delete image from local filesystem
 */
export async function deleteFromLocal(key: string): Promise<void> {
  const filepath = join(LOCAL_STORAGE_DIR, key)
  assertSafePath(filepath)

  try {
    await unlink(filepath)
  } catch {
    // File may not exist, ignore
  }
}
```

```typescript
import { env, isS3Configured } from '@/lib/env'
import { uploadWeaponImage, type UploadImageResult } from './upload-image'
import { deleteWeaponImage, extractKeyFromUrl } from './delete-image'
import { uploadToLocal, deleteFromLocal, type LocalUploadResult } from './local-storage'

export interface StorageUploadParams {
  weaponId: string
  imageData: Buffer
  assetId?: string
}

export interface StorageResult {
  key: string
  url: string
}

/**
 * Upload image to storage (S3 or local fallback in dev only)
 */
export async function uploadImage(params: StorageUploadParams): Promise<StorageResult> {
  if (isS3Configured()) {
    return uploadWeaponImage(params)
  }

  if (env.NODE_ENV === 'production') {
    throw new Error('S3 must be configured in production. Local storage fallback is not available.')
  }

  console.warn('S3 not configured, using local storage fallback')
  return uploadToLocal(params)
}

/**
 * Delete image from storage
 */
export async function deleteImage(urlOrKey: string): Promise<void> {
  if (isS3Configured()) {
    const key = urlOrKey.includes('://') ? extractKeyFromUrl(urlOrKey) : urlOrKey
    await deleteWeaponImage(key)
  } else {
    const key = urlOrKey.startsWith('/uploads/') ? urlOrKey.replace('/uploads/', '') : urlOrKey
    await deleteFromLocal(key)
  }
}

// Re-export types
export type { UploadImageResult, LocalUploadResult }
```

```typescript
import { existsSync } from 'fs'
import { rm } from 'fs/promises'
import { join } from 'path'
import { uploadToLocal, deleteFromLocal } from '../local-storage'

const TEST_DIR = './public/uploads/weapons'

describe('local-storage', () => {

    try {

    } catch {
      // Ignore cleanup errors
    }

        weaponId,
        imageData,
        assetId: 'test-asset',
      })

      expect(result.key).toBe('weapons/test-weapon-123/test-asset.png')
      expect(result.url).toBe('/uploads/weapons/test-weapon-123/test-asset.png')

      // Verify file exists
      const filepath = join('./public/uploads', result.key)
      expect(existsSync(filepath)).toBe(true)
    })

    it('generates asset ID if not provided', async () => {
      const imageData = Buffer.from('fake-png-data')
      const weaponId = 'test-weapon-456'

      const result = await uploadToLocal({
        weaponId,
        imageData,
      })

      expect(result.key).toMatch(/^weapons\/test-weapon-456\/\d+-[a-z0-9]+\.png$/)
    })

    it('rejects weaponId with path traversal characters', async () => {
      const imageData = Buffer.from('fake-png-data')
      await expect(uploadToLocal({
        weaponId: '../../../etc',
        imageData,
      })).rejects.toThrow('Invalid storage ID')
    })
  })

  describe('deleteFromLocal', () => {
    it('deletes existing file', async () => {
      const imageData = Buffer.from('fake-png-data')
      const weaponId = 'test-weapon-789'

      const result = await uploadToLocal({
        weaponId,
        imageData,
        assetId: 'to-delete',
      })

      const filepath = join('./public/uploads', result.key)
      expect(existsSync(filepath)).toBe(true)

      await deleteFromLocal(result.key)
      expect(existsSync(filepath)).toBe(false)
    })

    it('handles non-existent file gracefully', async () => {
      // Should not throw
      await expect(deleteFromLocal('weapons/fake/nonexistent.png')).resolves.toBeUndefined()
    })
  })
})
```
